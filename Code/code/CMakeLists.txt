cmake_minimum_required(VERSION 3.16)
cmake_policy(SET CMP0091 NEW)
project(second_hand_platform VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(market src/repository.cc src/security.cc src/service.cc src/collections.cc)
target_include_directories(market PUBLIC include)
if(MSVC)
  target_compile_options(market PUBLIC /W4 /WX /utf-8)
else()
  target_compile_options(market PUBLIC -Wall -Wextra -Wpedantic -Werror)
endif()
if(WIN32)
  target_link_libraries(market PUBLIC bcrypt)
else()
  find_package(OpenSSL REQUIRED)
  target_link_libraries(market PUBLIC OpenSSL::Crypto)
endif()
add_executable(market_cli src/main.cc)
target_link_libraries(market_cli PRIVATE market)
if(WIN32)
  enable_language(RC)
  add_executable(market_gui WIN32 src/gui_main.cc src/gui.cc
    src/gui_support.cc resources/market_gui.rc)
  target_compile_definitions(market_gui PRIVATE UNICODE _UNICODE NOMINMAX
    _WIN32_WINNT=0x0601 WINVER=0x0601)
  target_link_libraries(market_gui PRIVATE market comctl32 gdi32 user32 shell32)
  add_executable(gui_support_tests tests/gui_support_test.cc src/gui_support.cc)
  target_compile_definitions(gui_support_tests PRIVATE UNICODE _UNICODE NOMINMAX
    _WIN32_WINNT=0x0601 WINVER=0x0601)
  target_link_libraries(gui_support_tests PRIVATE market gdi32 user32)
endif()
enable_testing()
if(WIN32)
  add_test(NAME desktop_input_contracts COMMAND gui_support_tests)
endif()
add_executable(market_tests tests/market_test.cc)
target_link_libraries(market_tests PRIVATE market)
add_test(NAME business_and_persistence COMMAND market_tests)
add_executable(collections_tests tests/collections_test.cc)
target_link_libraries(collections_tests PRIVATE market)
add_test(NAME collections_and_migration COMMAND collections_tests)
set_tests_properties(collections_and_migration PROPERTIES TIMEOUT 120)
set_tests_properties(business_and_persistence PROPERTIES TIMEOUT 120)
if(MINGW)
  target_link_options(collections_tests PRIVATE -static -static-libgcc -static-libstdc++)
  target_link_options(gui_support_tests PRIVATE -static -static-libgcc -static-libstdc++)
  target_link_options(market_gui PRIVATE -municode -static -static-libgcc -static-libstdc++)
  target_link_options(market_cli PRIVATE -municode)
  target_link_options(market_cli PRIVATE -static -static-libgcc -static-libstdc++)
  target_link_options(market_tests PRIVATE -static -static-libgcc -static-libstdc++)
endif()
